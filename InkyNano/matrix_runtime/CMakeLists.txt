cmake_minimum_required(VERSION 3.16)
project(inky_prompt_matrix LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)

if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned Inkling-compatible llama.cpp checkout")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(inky-matrix runtime.cpp)
target_include_directories(inky-matrix PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/src")
target_link_libraries(inky-matrix PRIVATE llama ggml)
add_executable(inky-packed-expert packed_expert.cpp)
target_include_directories(inky-packed-expert PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(inky-packed-expert PRIVATE ggml)
if(WIN32)
    target_link_libraries(inky-matrix PRIVATE psapi)
endif()
